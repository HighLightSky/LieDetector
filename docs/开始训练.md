## 完整训练流程

### 步骤 1: 准备数据

```bash
# 如果还没有标签文件，先生成
python run.py
```

### 步骤 2: 提取特征

```bash
# 提取所有视频的特征（需要2-3小时）
python -c "from dataloader.precompute_features import main; main()"
```

这会在 `src/features/` 目录下生成：
- `{video_name}_faces.pt`: 人脸特征
- `{video_name}_audios.pt`: 音频特征
- `{video_name}_openfaces.pt`: OpenFace特征

### 步骤 3: 训练模型

```bash
# 基本训练（使用默认参数）
python train.py

# 自定义训练
python train.py \
    --batch_size 16 \
    --stage1_epochs 10 \
    --stage2_epochs 10 \
    --stage3_epochs 10 \
    --device cuda
```

训练完成后，模型保存在 `checkpoints/` 目录。

### 步骤 4: 使用训练好的模型

```python
from detect import MultiModalFusionModel, detect_video
from dataloader import FaceExtractor, AudioExtractor, OpenFaceExtractor
import torch

# 加载模型
device = 'cuda'
model = MultiModalFusionModel(device=device, use_openface=True)

# 加载训练好的权重
checkpoint = torch.load('checkpoints/best_stage3.pth')
model.load_state_dict(checkpoint['model_state_dict'])
model.eval()

# 创建提取器
face_extractor = FaceExtractor(num_frames=16)
audio_extractor = AudioExtractor(device=device)
openface_extractor = OpenFaceExtractor(num_frames=16)

# 检测视频
pred, probs = detect_video(
    video_path='path/to/video.mp4',
    model=model,
    face_extractor=face_extractor,
    audio_extractor=audio_extractor,
    openface_extractor=openface_extractor,
    device=device,
    verbose=True
)

print(f"预测: {'deception' if pred == 1 else 'truth'}")
print(f"置信度: {probs[pred]:.2%}")
```

```bash
python -c "
from dataloader import FaceExtractor, AudioExtractor, OpenFaceExtractor
import torch

face_ext = FaceExtractor(num_frames=16)
audio_ext = AudioExtractor(device='cuda')
openface_ext = OpenFaceExtractor(num_frames=16)

video_path = 'src/videos/cut/AN_WILTY_EP15_lie10.mp4'
faces = face_ext.extract_from_video(video_path, return_tensor=True)
audios = audio_ext.extract_from_video(video_path, return_tensor=True)
openfaces = openface_ext.extract_from_video(video_path, return_tensor=True)

torch.save(faces, 'faces.pt')
torch.save(audios, 'audios.pt')
torch.save(openfaces, 'openfaces.pt')
"
```

### 推理

```bash
# 测试推理
python detect.py

# 批量推理
python -c "
from detect import MultiModalFusionModel, detect_video
from dataloader import FaceExtractor, AudioExtractor, OpenFaceExtractor
from pathlib import Path
import torch

device = 'cuda'
model = MultiModalFusionModel(device=device, use_openface=True)
checkpoint = torch.load('checkpoints/best_stage3.pth')
model.load_state_dict(checkpoint['model_state_dict'])

face_ext = FaceExtractor(num_frames=16)
audio_ext = AudioExtractor(device=device)
openface_ext = OpenFaceExtractor(num_frames=16)

video_dir = Path('src/videos/cut')
for video_path in list(video_dir.glob('*.mp4'))[:10]:
    pred, probs = detect_video(
        video_path, model, face_ext, audio_ext, openface_ext, device, verbose=False
    )
    print(f'{video_path.name}: {\"deception\" if pred == 1 else \"truth\"} ({probs[pred]:.2%})')
"
```

## 故障排除

### 问题: ModuleNotFoundError

```bash
# 确保在项目根目录
cd /path/to/lie_detector

# 检查Python路径
python -c "import sys; print(sys.path)"
```

### 问题: CUDA out of memory

```bash
# 减小批大小
python train.py --batch_size 8

# 或使用CPU
python train.py --device cpu
```

### 问题: 特征文件不存在

```bash
# 先提取特征
python -c "from dataloader.precompute_features import main; main()"
```

### 问题: OpenFace不可用

```bash
# 跳过OpenFace
python train.py --use_openface False
python detect.py  # 会自动跳过OpenFace
```

### 混合精度训练

在 `trainer/trainer.py` 中添加：
```python
from torch.cuda.amp import autocast, GradScaler

scaler = GradScaler()
with autocast():
    logits = self.model(faces, audios, openfaces)
    loss = criterion(logits, labels)
scaler.scale(loss).backward()
scaler.step(optimizer)
scaler.update()
```
